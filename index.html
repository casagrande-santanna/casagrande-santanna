-- ===================================================================== -- LIMITES DE RESERVA NO SERVIDOR -- Passa a valer de verdade (e não só na tela) o que está em -- Config > Regras de reserva: -- * Reservas pontuais abertas ao mesmo tempo (por apto) -> quadra -- * Antecedência mínima para reservar (minutos) -> quadra -- * Antecedência mínima no fim de semana (horas) -> quadra -- * Antecedência máxima para reservar (dias) -> quadra e academia -- Administrador e o uso imediato pelo QR Code da quadra não são limitados. -- Cole TUDO no SQL Editor e clique em Run. Pode rodar mais de uma vez. -- ===================================================================== BEGIN; ALTER TABLE public.configuracoes ADD COLUMN IF NOT EXISTS max_reservas_simult int DEFAULT 2; ALTER TABLE public.configuracoes ADD COLUMN IF NOT EXISTS antecedencia_min_h int DEFAULT 60; -- em MINUTOS (nome antigo) ALTER TABLE public.configuracoes ADD COLUMN IF NOT EXISTS antecedencia_max_dias int DEFAULT 15; ALTER TABLE public.configuracoes ADD COLUMN IF NOT EXISTS ant_fds_h int DEFAULT 0; -- em HORAS CREATE OR REPLACE FUNCTION public.reservas_limites() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE agora timestamp := agora_local(); v_ini timestamp := NEW.data_reserva + NEW.slot_inicio::time; v_quadra boolean; v_qr boolean := coalesce(NEW.checkin_feito, false); v_min int := cfg_int('antecedencia_min_h', 60); v_fds int := cfg_int('ant_fds_h', 0); v_maxd int := cfg_int('antecedencia_max_dias', 15); v_sim int := cfg_int('max_reservas_simult', 2); v_abertas int; BEGIN IF NOT is_app_user() OR is_sistema() OR is_admin() THEN RETURN NEW; END IF; v_quadra := NOT EXISTS (SELECT 1 FROM espacos e WHERE e.id = NEW.espaco_id AND e.nome ILIKE '%academia%'); IF v_maxd > 0 AND NEW.data_reserva > agora::date + v_maxd THEN RAISE EXCEPTION 'Reservas só podem ser feitas com até % dias de antecedência.', v_maxd; END IF; IF NOT v_quadra OR v_qr THEN RETURN NEW; END IF; IF v_min > 0 AND v_ini < agora + make_interval(mins => v_min) THEN RAISE EXCEPTION 'Reservas da quadra precisam de pelo menos % minutos de antecedência.', v_min; END IF; IF v_fds > 0 AND extract(dow FROM NEW.data_reserva) IN (0, 6) AND v_ini < agora + make_interval(hours => v_fds) THEN RAISE EXCEPTION 'No fim de semana, reserve com pelo menos % hora(s) de antecedência.', v_fds; END IF; IF v_sim > 0 THEN -- Trava por apartamento: dois pedidos ao mesmo tempo não furam o limite PERFORM pg_advisory_xact_lock(hashtext('limite-apto-' || NEW.unidade_id::text)); SELECT count(*) INTO v_abertas FROM reservas r LEFT JOIN espacos e ON e.id = r.espaco_id WHERE r.unidade_id = NEW.unidade_id AND r.status = 'confirmada' AND coalesce(r.tipo_reserva, 'pontual') = 'pontual' AND coalesce(e.nome, '') NOT ILIKE '%academia%' AND r.data_reserva + r.slot_fim::time > agora; IF v_abertas >= v_sim THEN RAISE EXCEPTION 'Seu apartamento já tem % reserva(s) da quadra em aberto. Use ou cancele uma antes de reservar outra.', v_abertas; END IF; END IF; RETURN NEW; END; $$; -- O nome faz este gatilho rodar logo depois de trg_reservas_regras DROP TRIGGER IF EXISTS trg_reservas_regras_limites ON public.reservas; CREATE TRIGGER trg_reservas_regras_limites BEFORE INSERT ON public.reservas FOR EACH ROW EXECUTE FUNCTION public.reservas_limites(); -- Atualizações em tempo real da tela (reservas, bloqueios e agenda do Google). -- O Realtime respeita o RLS: cada um só recebe o que já pode ver. DO $$ DECLARE t text; BEGIN FOREACH t IN ARRAY ARRAY['reservas', 'bloqueios', 'agenda_google'] LOOP IF to_regclass('public.' || t) IS NOT NULL AND NOT EXISTS ( SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND schemaname = 'public' AND tablename = t ) THEN EXECUTE format('ALTER PUBLICATION supabase_realtime ADD TABLE public.%I', t); END IF; END LOOP; END $$; COMMIT; NOTIFY pgrst, 'reload schema'; -- Conferência SELECT max_reservas_simult, antecedencia_min_h AS antecedencia_min_minutos, antecedencia_max_dias, ant_fds_h AS antecedencia_fds_horas FROM public.configuracoes LIMIT 1; SELECT tablename AS tabelas_em_tempo_real FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND schemaname = 'public';
